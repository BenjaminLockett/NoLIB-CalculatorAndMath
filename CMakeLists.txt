cmake_minimum_required(VERSION 3.15)
project(NoLibMath LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS Source/NoLib/*.cpp)
file(GLOB_RECURSE HEADERS CONFIGURE_DEPENDS Header/*.hpp)
set(WORKBENCH_SOURCES Source/Workbench/ExpressionParser.cpp Source/Workbench/Commands.cpp)

add_library(NoLibCore STATIC ${CORE_SOURCES})
target_include_directories(NoLibCore PUBLIC Header)
add_library(NoLibWorkbench STATIC ${WORKBENCH_SOURCES})
target_link_libraries(NoLibWorkbench PUBLIC NoLibCore)
add_executable(NoLibMath Source/main.cpp Source/Workbench/Console.cpp ${HEADERS})
target_link_libraries(NoLibMath PRIVATE NoLibWorkbench)
add_executable(NoLibMathTests Source/Tests/ParserTests.cpp)
target_link_libraries(NoLibMathTests PRIVATE NoLibWorkbench)

foreach(target NoLibCore NoLibWorkbench NoLibMath NoLibMathTests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
        if(NOT target MATCHES "^NoLib(Core|Workbench)$")
            target_link_options(${target} PRIVATE /STACK:4194304)
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
source_group(TREE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${HEADERS} ${CORE_SOURCES} ${WORKBENCH_SOURCES})
enable_testing()
add_test(NAME parser_and_commands COMMAND NoLibMathTests)
add_test(NAME command_line_example COMMAND NoLibMath --plain --command "action[dirive(x^3 + 2x + 1)]")
set_tests_properties(command_line_example PROPERTIES PASS_REGULAR_EXPRESSION "3x\\^2 \\+ 2")
add_test(NAME malformed_command COMMAND NoLibMath --plain --command "action[derive(x^)]")
set_tests_properties(malformed_command PROPERTIES WILL_FAIL TRUE)
